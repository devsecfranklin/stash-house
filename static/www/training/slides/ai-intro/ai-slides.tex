\section{Deterministic Controls \& Structured Outputs}

\begin{frame}[fragile]{Deterministic Sampling vs.\ Stochastic Generation}
 \begin{itemize}
 \item \textbf{Temperature ($T$):} Scales the logits before softmax.
 \begin{itemize}
 \item $T \to 0$: Argmax sampling / greedy decoding (reproducible classification, code, data extraction).
 \item $T \ge 0.7$: High variance and exploratory token distributions.
 \end{itemize}
 \item \textbf{Top-$P$ (Nucleus Sampling):} Dynamically pools smallest token set whose cumulative probability exceeds $P$.
 \item \textbf{Seed Calibration:} Fixes pseudo-random sampling seeds across deterministic API requests.
 \end{itemize}
\end{frame}

\begin{frame}[fragile]{JSON Mode vs.\ Constrained Grammars}
 \begin{itemize}
 \item \textbf{Prompt-Coaxed JSON (Unreliable):}
 \begin{itemize}
 \item Relies solely on system prompt instructions (e.g., \textit{"Respond only in valid JSON"}).
 \item Prone to syntax breakage, preamble comments, and schema hallucinations.
 \end{itemize}
 \item \textbf{Constrained Grammar Decoding (Reliable):}
 \begin{itemize}
 \item Decoders enforce regular expressions or context-free grammars (GBNF, JSON Schema) directly against logits.
 \item Tokens that violate the schema receive $-\infty$ logits and cannot be sampled.
 \end{itemize}
 \end{itemize}
\end{frame}

\end{frame}

 \begin{block}{Objective}
 Extract structured alerts from unstructured kernel trace lines using strict schema enforcement.
 \end{block}
 \begin{enumerate}
 \item Define a Pydantic schema extracting: \texttt{timestamp}, \texttt{service}, and \texttt{failure\_reason}.
 \item Configure client inference with \texttt{temperature=0.0} and strict JSON schema decoding.

 \end{enumerate}
\end{frame}

\begin{frame}{Summary}
 \begin{block}{Key Take-Aways}
 \begin{itemize}
 \item Deterministic sampling ($T=0$) enables reproducible, pipeline-safe outputs.
 \item Constrained grammar decoding is the only reliable path to schema-compliant JSON.
 \item Pydantic + structured-output APIs combine type-hint validation with LLM reasoning.
 \item Every production workflow should validate against $100\%$ parse-success targets.
 \end{itemize}
 \end{block}
